pg_get_viewdef
 WITH trials AS (
         SELECT events.session_id,
            events.rig_id,
            events.value::bigint AS trial,
            min(events.t_us) AS t_us
           FROM events
          WHERE events.type = 'STATE'::text AND events.t_us > 0
          GROUP BY events.session_id, events.rig_id, (events.value::bigint)
        )
 SELECT t.session_id,
    t.rig_id,
    t.trial,
    to_timestamp((t.t_us::numeric / 1000000.0)::double precision) AS trial_start,
    p.type AS param,
    p.value
   FROM trials t
     CROSS JOIN LATERAL ( SELECT DISTINCT ON (e.type) e.type,
            e.value
           FROM events e
          WHERE e.type ~~ 'PARAM%'::text AND e.session_id = t.session_id AND e.rig_id = t.rig_id AND e.t_us > 0 AND e.t_us <= t.t_us
          ORDER BY e.type, e.t_us DESC) p;
